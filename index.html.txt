    let currentCardCount = 3;
    let allowReversed = false;

    function toggleCategory(element) {
        document.querySelectorAll('.category-card').forEach(card => {
            if (card !== element) card.classList.remove('active');
        });
        element.classList.toggle('active');
        document.getElementById('bottomSheet').classList.remove('open');
    }

    function openBottomSheet(category, title, desc, count) {
        document.getElementById('bsCategory').innerText = category;
        document.getElementById('bsTitle').innerText = title;
        document.getElementById('bsDesc').innerText = desc;
        document.getElementById('bsCardCount').innerText = count;
        currentCardCount = count;
        document.getElementById('bottomSheet').classList.add('open');
    }

    function selectRadio(element, isReversed) {
        let siblings = element.parentElement.querySelectorAll('.radio-box');
        siblings.forEach(sibling => sibling.classList.remove('selected'));
        element.classList.add('selected');
        allowReversed = isReversed;
    }

    // 🔒 Web Crypto API 기반의 암호학적 난수 생성 및 셔플 함수
    function getCryptoShuffledDeck() {
        let deck = [...TAROT_DECK];
        for (let i = deck.length - 1; i > 0; i--) {
            const array = new Uint32Array(1);
            window.crypto.getRandomValues(array);
            const j = array[0] % (i + 1);
            [deck[i], deck[j]] = [deck[j], deck[i]];
        }
        return deck;
    }

    // 카드 펼치기 및 리딩 시작
    function startReading() {
        document.getElementById('bottomSheet').classList.remove('open');
        const modal = document.getElementById('spreadModal');
        const cardsGrid = document.getElementById('cardsGrid');
        const resultsContainer = document.getElementById('readingResults');
        
        cardsGrid.innerHTML = '';
        resultsContainer.innerHTML = '';
        
        const qText = document.getElementById('userQuestion').value.trim();
        document.getElementById('modalQuestionText').innerText = qText ? `"${qText}"` : "카드를 차례대로 터치해 보세요";

        const shuffled = getCryptoShuffledDeck();
        const selectedCards = shuffled.slice(0, currentCardCount);

        selectedCards.forEach((card, index) => {
            // 역방향 여부 난수 결정
            const randArray = new Uint8Array(1);
            window.crypto.getRandomValues(randArray);
            const isReversed = allowReversed && (randArray[0] % 2 === 1);

            const cardWrapper = document.createElement('div');
            cardWrapper.className = 'tarot-card-wrapper';
            cardWrapper.innerHTML = `
                <div class="tarot-card ${isReversed ? 'reversed' : ''}" id="card-${index}">
                    <div class="card-back">
                        <span>🌙</span>
                        <span style="font-size:10px; margin-top:4px; opacity:0.7;">LUNEA</span>
                    </div>
                    <div class="card-front">
                        <img src="${card.img}" alt="${card.name}">
                    </div>
                </div>
            `;

            cardWrapper.onclick = () => {
                const cardElem = document.getElementById(`card-${index}`);
                if (!cardElem.classList.contains('flipped')) {
                    cardElem.classList.add('flipped');
                    
                    // 결과 텍스트 추가
                    const infoBox = document.createElement('div');
                    infoBox.className = 'card-info-box';
                    infoBox.innerHTML = `
                        <h4>Card ${index + 1}: ${card.name} ${isReversed ? '(역방향 ↺)' : '(정방향)'}</h4>
                        <p><strong>핵심 상징:</strong> ${card.keyword}</p>
                    `;
                    resultsContainer.appendChild(infoBox);
                }
            };

            cardsGrid.appendChild(cardWrapper);
        });

        modal.classList.add('show');
    }

    function closeModal() {
        document.getElementById('spreadModal').classList.remove('show');
    }

    // 팝업 외부 닫기
    document.addEventListener('click', function(e) {
        const bs = document.getElementById('bottomSheet');
        if (!bs.contains(e.target) && !e.target.closest('.reading-item') && bs.classList.contains('open')) {
            bs.classList.remove('open');
        }
    });
